cmake_minimum_required(VERSION 3.16)
project(nav_obstacle_map)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
ament_python_install_package(${PROJECT_NAME} PACKAGE_DIR python/${PROJECT_NAME})
install(PROGRAMS python/nav_obstacle_map/mapper_node.py DESTINATION lib/${PROJECT_NAME} RENAME student_mapper)
install(PROGRAMS python/nav_obstacle_map/prior_node.py DESTINATION lib/${PROJECT_NAME} RENAME global_prior)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/launch")
 install(DIRECTORY launch DESTINATION share/${PROJECT_NAME})
endif()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/config")
 install(DIRECTORY config DESTINATION share/${PROJECT_NAME})
endif()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/resource")
 install(DIRECTORY resource DESTINATION share/${PROJECT_NAME})
endif()
ament_package()
