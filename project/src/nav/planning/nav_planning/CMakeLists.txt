cmake_minimum_required(VERSION 3.16)
project(nav_planning)
find_package(ament_cmake REQUIRED)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra)
find_package(rclcpp REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_msgs REQUIRED)
add_library(legacy_dstar src/legacy_dstar.cpp)
target_include_directories(legacy_dstar PUBLIC include)
add_executable(planner_node src/planner_node.cpp)
target_link_libraries(planner_node legacy_dstar)
ament_target_dependencies(planner_node rclcpp nav_msgs geometry_msgs)
install(TARGETS planner_node DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
 enable_testing()
 add_executable(test_legacy tests/test_legacy.cpp)
 target_link_libraries(test_legacy legacy_dstar)
 add_test(NAME legacy_dstar_math COMMAND test_legacy)
endif()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/launch")
 install(DIRECTORY launch DESTINATION share/${PROJECT_NAME})
endif()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/config")
 install(DIRECTORY config DESTINATION share/${PROJECT_NAME})
endif()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/resource")
 install(DIRECTORY resource DESTINATION share/${PROJECT_NAME})
endif()
ament_package()
