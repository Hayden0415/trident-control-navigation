cmake_minimum_required(VERSION 3.16)
project(nav_legacy_control)
find_package(ament_cmake REQUIRED)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra)
find_package(rclcpp REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_msgs REQUIRED)
add_executable(controller_node src/controller_node.cpp)
target_include_directories(controller_node PRIVATE include)
ament_target_dependencies(controller_node rclcpp nav_msgs geometry_msgs std_msgs)
install(TARGETS controller_node DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
 enable_testing()
 add_executable(test_legacy_pid tests/test_legacy_pid.cpp)
 target_include_directories(test_legacy_pid PRIVATE include)
 add_test(NAME legacy_pid_math COMMAND test_legacy_pid)
endif()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/launch")
 install(DIRECTORY launch DESTINATION share/${PROJECT_NAME})
endif()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/config")
 install(DIRECTORY config DESTINATION share/${PROJECT_NAME})
endif()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/resource")
 install(DIRECTORY resource DESTINATION share/${PROJECT_NAME})
endif()
ament_package()
